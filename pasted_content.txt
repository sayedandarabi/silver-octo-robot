        <div class="t"><b>${s.name}</b><span>${s.phone}</span></div>
        <div class="amt ${s.due?'down':''}">${s.due?money(s.due):'تسویه'}</div>
      </div>`).join('');
  }
  document.addEventListener('click', e=>{
    const b = e.target.closest('[data-pay]');
    if(!b) return;
    const c = S.customers.find(x=>x.id===+b.dataset.pay);
    const amt = Math.min(c.debt, 10000);
    if(!amt){ toast('مانده‌ای ندارد'); return; }
    c.debt -= amt; S.cash += amt;
    S.feed.unshift({t:'دریافت پول از مشتری', s:c.name+' · صندوق اصلی', a:amt, ok:true});
    S.feed = S.feed.slice(0,6);
    save(); renderHome(); renderCust();
    toast('دریافت ثبت شد: ' + money(amt));
  });

  /* ---------- expenses ---------- */
  const cats = ['کرایه','برق','آب','انترنت','ترانسپورت','متفرقه'];
  $('#expChips').innerHTML = cats.map((c,i)=>`<button class="chip ${i?'':'on'}">${c}</button>`).join('');
  document.addEventListener('click', e=>{
    if(e.target.classList.contains('chip') && e.target.parentElement.id === 'expChips'){
      [...e.target.parentElement.children].forEach(x=>x.classList.remove('on'));
      e.target.classList.add('on');
    }
  });
  $('#saveExp').onclick = ()=>{
    const v = +($('#expAmt').value || '').replace(/\D/g,'');
    if(!v){ toast('مبلغ را وارد کنید'); return; }
    S.cash -= v;
    S.feed.unshift({t:'ثبت مصرف', s:'دسته: '+(document.querySelector('#expChips .chip.on')?.textContent||'—'), a:v, ok:false});
    S.feed = S.feed.slice(0,6);
    save(); renderHome(); $('#expAmt').value = '';
    toast('مصرف ثبت شد');
    go('home');
  };

  /* ---------- reports ---------- */
  const data7 = [6, 9, 5, 11, 8, 14, 10];
  const data30 = [42, 55, 38, 61, 47, 70, 58];
  const data90 = [120, 168, 141, 190, 152, 205, 178];
  let days = 7;
  function drawChart(){
    const d = days === 7 ? data7 : days === 30 ? data30 : data90;
    const max = Math.max(...d);
    const labels = days === 7 ? ['شن','یک','دو','سه','چهار','پنج','جمعه'] : ['هفته ۱','هفته ۲','هفته ۳','هفته ۴','هفته ۵','هفته ۶','هفته ۷'];
    $('#chart').innerHTML = d.map((v,i)=>`<div class="bar-wrap"><div class="bar" data-h="${Math.round(v/max*100)}"></div><span>${labels[i]}</span></div>`).join('');
    requestAnimationFrame(()=>document.querySelectorAll('.bar').forEach(b=>b.style.height = b.dataset.h + '%'));
    const sum = d.reduce((a,b)=>a+b,0) * 1000;
    $('#repSum').textContent = money(sum);
    $('#repProfit').textContent = money(sum * 0.23);
  }
  document.querySelectorAll('#repChips .chip').forEach(c=>c.onclick = ()=>{
    days = +c.dataset.days;
    document.querySelectorAll('#repChips .chip').forEach(x=>x.classList.toggle('on', x===c));
    drawChart();
  });

  /* ---------- ledger ---------- */
  function renderLedger(){
    $('#ledgerList').innerHTML = S.ledger.map(l=>`
      <div class="row">
        <div class="t"><b>${l.acc}</b><span>حساب کنترلی</span></div>
        <div style="text-align:end;font-size:11px">
          <div class="up">بدهکار ${money(l.dr)}</div>
          <div class="down">بستانکار ${money(l.cr)}</div>
        </div>
      </div>`).join('') + `<div class="note" style="margin-top:8px">جمع بدهکار و بستانکار برابر است — سند نامتوازن رد می‌شود.</div>`;
  }

  /* ---------- backup ---------- */
  $('#backupBtn').onclick = ()=>{
    const b = btoa(unescape(encodeURIComponent(JSON.stringify(S))));
    toast('نسخه پشتیبان رمزنگاری‌شده ساخته شد · ' + fd(b.length) + ' بایت');
  };

  /* ---------- clock ---------- */
  setInterval(()=>{
    const t = new Date();
    $('#clock').textContent = fd(String(t.getHours()).padStart(2,'0')) + ':' + fd(String(t.getMinutes()).padStart(2,'0'));
  }, 1000);

  /* ---------- init ---------- */
  renderHome(); renderChips(); renderCart(); renderStock(); renderCust(); renderLedger(); drawChart();
  $('#tabbar').classList.add('hide');
})();
